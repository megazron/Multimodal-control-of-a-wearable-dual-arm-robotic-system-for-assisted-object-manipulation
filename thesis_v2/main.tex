%% Implementation provenance. Every claim about the implementation names the

%% Repository paths are long and unbreakable in typewriter type, which
%% produces overfull lines in captions and narrow columns. Underscores are
%% given a zero-width break opportunity so a path can wrap.

  \def\_{\char`\_\hspace{0pt}}\def\slash{/\hspace{0pt}}#1}}
\DeclareRobustCommand{\file}[1]{\repo{#1}}

%% ---------------------------------------------------------------------------
%% RESERVED SPACE. An image that does not exist yet is drawn as a marked box
%% rather than omitted, so a missing figure is visible in the compiled
%% document instead of silently absent. #1 is the height, #2 what the image
%% must show.
%% ---------------------------------------------------------------------------
\newcommand{\imgplaceholder}[2]{%
  \begin{tikzpicture}
    \node[draw, dashed, thick, red!60!black, fill=red!3,
          minimum height=#1, text width=0.86\linewidth, align=center,
          inner sep=10pt] (ph) {%
      {\bfseries\color{red!60!black} PLACEHOLDER: IMAGE REQUIRED}\\[4pt]
      \small\color{black}#2};
  \end{tikzpicture}}

%% Screenshots of the simulator, the operating window or a live view.
\newcommand{\screenshot}[2]{%
  \begin{tikzpicture}
    \node[draw, dashed, thick, blue!45!black, fill=blue!3,
          minimum height=#1, text width=0.86\linewidth, align=center,
          inner sep=10pt]{%
      {\bfseries\color{blue!45!black} SPACE RESERVED FOR A SCREENSHOT}\\[5pt]
      \small\color{black}#2};
  \end{tikzpicture}}

%% A figure reproduced from another publication. Permission is required
%% before submission, so the requirement travels with the box.
\newcommand{\reprodplaceholder}[2]{%
  \begin{tikzpicture}
    \node[draw, dashed, thick, orange!70!black, fill=orange!4,
          minimum height=#1, text width=0.86\linewidth, align=center,
          inner sep=10pt] {%
      {\bfseries\color{orange!70!black} PLACEHOLDER: REPRODUCED FIGURE}\\[4pt]
      \small\color{black}#2\\[4pt]
      {\footnotesize\itshape Reproduction requires written permission from the
      copyright holder. Not yet obtained.}};
  \end{tikzpicture}}
