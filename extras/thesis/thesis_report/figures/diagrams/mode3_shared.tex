% MODE 4 IN THE REGISTRY -- SHARED AUTONOMY.
% Drawn so the position path never passes through the autonomy stage. That
% separation is the safety argument of the whole design, so it is the one
% thing the figure has to make unmissable.
\begin{tikzpicture}[x=32mm, y=14mm]

\node[sensor] (master) at (0, 3.4) {master arm\\(mode 1 front end)};
\node[proc]   (pose)   at (1, 3.4) {commanded pose\\position $+$ fixed wrist};

\node[sensor] (cam)    at (0, 0.5) {wrist cameras\\RGB-D};
\node[proc]   (tag)    at (1, 0.5) {AprilTag 36h11\\\texttt{SUBPIX} refinement};
\node[proc]   (track)  at (2, 0.5) {object tracker\\\emph{drops} at \SI{1.5}{\second}\\unseen};
\node[proc]   (grasp)  at (3, 0.5) {grasp generator\\every candidate\\IK-validated};

\node[proc, text width=31mm] (intent) at (2, 2.1)
      {intent estimator\\pointing (primary),\\velocity (weight 0.25),\\forgetting $\lambda = 0.02$};
\node[decision] (amb)  at (3.15, 2.1) {ambiguous?};
\node[safety, text width=31mm] (arb) at (4.4, 2.1)
      {arbiter\\DIRECT $\rightarrow$ ASSIST\\requires $P > \theta$,\\proximity, and\\not ambiguous};

\node[proc, fill=green!6, draw=green!45!black, text width=32mm] (posn) at (5.6, 3.6)
      {\textbf{position: 100\% the}\\\textbf{operator's, in every state}};
\node[proc]   (slerp)  at (5.6, 1.5) {wrist SLERP\\over \SI{0.5}{\second}};

\node[safety] (ik)     at (6.7, 2.5) {collision-aware IK\\clearance floor\\step guard};
\node[actuator](arm)   at (7.6, 2.5) {Kinova Gen3};

\node[proc, fill=red!4, draw=red!50!black, text width=32mm] (stay) at (3.15, -1.0)
      {\textbf{stay in DIRECT}\\assist nothing};

\draw[flow] (master) -- (pose);
\draw[flow] (cam) -- (tag);
\draw[flow] (tag) -- (track);
\draw[flow] (track) -- (grasp);
\draw[flow] (track) -- (intent);
\draw[flow] (pose) -- (intent);
\draw[flow] (intent) -- (amb);
\draw[flow] (amb) -- node[lbl, above] {no} (arb);
\draw[flow] (grasp.east) to[out=0,in=250] (arb.south);
\draw[flow] (arb) -- (slerp);
\draw[flow] (pose.east) to[out=0,in=180] (posn.west);
\draw[flow] (slerp.east) to[out=0,in=250] (ik.south west);
\draw[flow] (posn.east) to[out=0,in=110] (ik.north west);
\draw[flow] (ik) -- (arm);

\draw[fail] (amb.south) -- node[faillbl, right, pos=0.35] {yes} (stay.north);
\draw[fail] (track.south) to[out=270,in=180] node[faillbl, below, pos=0.35]
      {no objects} (stay.west);

\node[font=\scriptsize\itshape, anchor=north west, text width=200mm] at (-0.45, -2.0)
  {The green path carries position and never enters the autonomy stage, so no
   failure of intent estimation, object tracking or grasp generation can move
   the arm. Cancelling freezes the orientation blend where it is rather than
   snapping it back, which keeps the commanded pose continuous. The gripper is
   never closed by this mode. A tie between two adjacent objects holds
   $P = 0.5$ exactly and the arbiter stays in DIRECT, because assisting toward
   the wrong one of two candidates is worse than not assisting.};
\end{tikzpicture}
