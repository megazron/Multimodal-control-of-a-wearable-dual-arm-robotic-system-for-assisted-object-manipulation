\begin{tikzpicture}[x=1mm, y=1mm, font=\scriptsize,

  rec/.style   = {sensor, text width=26mm, minimum height=9mm},
  halt/.style  = {safety, text width=26mm, minimum height=9mm}]

\node[pers] (consent) at (0, 0)   {information sheet,\\consent (SETREC 7111986)};

\node[stp] (assist)  at (93, 0)  {session: with assistance\\same task, same operator};

\node[stp] (metrics) at (93, -19) {metrics computed from the log:\\time, robot path, re-grips, lag};

\draw[flow] (assist.south) to[out=250,in=60] (log.north east);
