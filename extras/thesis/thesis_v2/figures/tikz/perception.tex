% Measure the room, then plan against what was measured.
\begin{tikzpicture}[x=1mm, y=1mm, font=\scriptsize]

  \node[stage, minimum height=9mm, text width=34mm] (sweep) at (0,0)
    {\textbf{1. Look}\\ the arm walks a grid of viewpoints, holding the wrist
     still and moving in straight rows};
  \node[stage, minimum height=9mm, text width=34mm, right=9mm of sweep] (seg)
    {\textbf{2. Cut up the picture}\\ every region in each frame is separated
     and lifted using that frame's own depth};
  \node[stage, minimum height=9mm, text width=34mm, right=9mm of seg] (fuse)
    {\textbf{3. Fuse}\\ many views become one map: a measured surface and the
     objects standing on it};

  \draw[go] (sweep) -- (seg); \draw[go] (seg) -- (fuse);

  \node[stage, minimum height=9mm, text width=34mm, below=10mm of fuse] (world)
    {\textbf{4. Hand the map to the planner}\\ a point on the table is now an
     obstacle; a point above it is not};
  \node[stage, minimum height=9mm, text width=34mm, left=9mm of world] (plan)
    {\textbf{5. Plan a path}\\ straight line first; search around the obstacle
     only if the straight line fails};
  \node[motor, minimum height=9mm, text width=34mm, left=9mm of plan] (exec)
    {\textbf{6. Execute}\\ approach, grasp, lift, carry, place, retreat};

  \draw[go] (fuse) -- (world);
  \draw[go] (world) -- (plan); \draw[go] (plan) -- (exec);

  \node[guard, minimum height=8mm, text width=76mm, below=9mm of plan] (stale)
    {\textbf{Refuses to move on a map older than thirty minutes.} Planning
     from an old map is harmless; driving the arm at coordinates measured in a
     previous session is not, and a stale file looks exactly as authoritative
     as a fresh one.};

  \node[tag, text width=104mm, align=left, below=5mm of stale] (note)
    {Before this existed, every coordinate the arm used was written in a file
     by hand. All of it was verified, and none of it survived a different
     table, a different object, or the same object moved fifty millimetres.};

\end{tikzpicture}
