%% FONT GUARD. The stock template hard-codes Times New Roman, which is a
%% Microsoft font and absent on most Linux machines, so the document failed to
%% build anywhere else. TeX Gyre Termes is a metric-compatible clone shipped
%% with TeX Live, so the fallback is visually equivalent.
\IfFontExistsTF{Times New Roman}{
  \setmainfont{Times New Roman}
}{
  \IfFontExistsTF{TeX Gyre Termes}{
    \setmainfont{TeX Gyre Termes}
  }{
    \setmainfont{texgyretermes}[
      Extension = .otf,
      UprightFont = *-regular,
      BoldFont = *-bold,
      ItalicFont = *-italic,
      BoldItalicFont = *-bolditalic]
  }
}
