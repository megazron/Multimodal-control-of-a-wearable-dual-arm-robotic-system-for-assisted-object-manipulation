% MODE 01 -- DIRECT TELEOPERATION FROM THE MANNEQUIN MASTER.
%
% Drawn as a VERTICAL spine because the four mode figures are meant to be
% compared stage by stage, and a horizontal chain of eleven stages does not
% fit a portrait page: the earlier landscape version had to be scaled to
% \textheight, which collided the failure labels with the boxes they
% annotate. The grid is shared by all four mode figures:
%
%     x = -52 mm   what happens when a stage refuses  (dashed red, leftwards)
%     x =   0 mm   the command spine, top to bottom
%     x = +52 mm   guards and side inputs             (solid, rightwards)
%
\begin{tikzpicture}[x=1mm, y=1mm, font=\scriptsize,
  spine/.style   = {proc,   text width=40mm},
  guardn/.style  = {safety, text width=38mm},
  sensen/.style  = {sensor, text width=30mm},
  sinkn/.style   = {draw, dashed, rounded corners=2pt, red!60!black,
                    fill=red!3, text width=34mm, align=center,
                    font=\scriptsize, inner sep=3pt},
  outn/.style    = {actuator, text width=40mm}]

% ---- sensing ---------------------------------------------------------
\node[sensen] (pots) at (-46,  10) {14 potentiometers\\12-bit ADC};
\node[sensen] (imu)  at (  0,  10) {2 $\times$ MPU-6050\\accelerometer, gyroscope};
\node[sensen] (btn)  at ( 46,  10) {2 buttons, 2 FSRs\\clutch and grip};

% ---- the spine -------------------------------------------------------
\node[spine] (teensy) at (0, -10) {Teensy 4.1\\exponential smoothing, clamp\\frame at \SI{100}{\hertz}};
\node[spine] (serial) at (0, -28) {USB serial\\exclusive \texttt{flock}};
\node[spine] (chan)   at (0, -46) {channel manager\\one verdict per channel,\\over \emph{distinct} updates};
\node[guardn](valid)  at (0, -64) {frame validation\\scoped to the channels the\\active mode consumes};
\node[spine] (sph)    at (0, -84) {spherical decomposition\\$\theta\!\leftarrow\!$ gravity, $\phi\!\leftarrow\!$ j1,
                                   $r\!\leftarrow\!$ FK};
\node[spine] (scale)  at (0,-104) {scale and anchor\\$\mathbf{p}=\mathbf{p}_0+s(\mathbf{d}-\mathbf{d}_{\mathrm{ref}})$};
\node[guardn](ik)     at (0,-124) {collision-aware IK\\redundancy re-seeded on refusal};
\node[guardn](guard)  at (0,-144) {step guard\\slews, never discards};
\node[outn]  (traj)   at (0,-162) {joint trajectory controller\\Kinova Gen3, \SI{100}{\hertz}};

% ---- guards and side inputs, to the right ----------------------------
\node[spine, text width=36mm] (degr)  at (62, -46) {degraded mode\\freeze incoherent channels,\\j7 radial fallback};
\node[spine, text width=36mm] (clutch)at (62, -84) {clutch\\quasi-static gate,\\anchor re-latch};
\node[guardn,text width=36mm] (estop) at (62,-104) {emergency stop\\dead-man on the \emph{source}\\timestamp};
\node[guardn,text width=36mm] (floor) at (62,-124) {clearance floor\\\SI{150}{\milli\metre} to the wearer,\\measured geometrically};

% ---- refusal sinks, to the left --------------------------------------
\node[sinkn] (hold)  at (-62, -64) {invalid frame:\\\textbf{hold the last pose}};
\node[sinkn] (nosol) at (-62,-124) {no solution:\\\textbf{hold, and say so}};
\node[sinkn] (park)  at (-62,-104) {latched:\\\textbf{park, then talk}};

\draw[flow] (pots) -- (teensy);
\draw[flow] (imu)  -- (teensy);
\draw[flow] (btn)  -- (teensy);
\foreach \a/\b in {teensy/serial, serial/chan, chan/valid, valid/sph,
                   sph/scale, scale/ik, ik/guard, guard/traj}
  {\draw[flow] (\a) -- (\b);}
\draw[flow] (degr.west)   -- (chan.east);
\draw[flow] (clutch.west) -- (sph.east);
\draw[flow] (estop.west)  -- (scale.east);
\draw[flow] (floor.west)  -- (ik.east);
\draw[fail] (valid.west) -- (hold.east);
\draw[fail] (ik.west)    -- (nosol.east);
\draw[fail] (estop.south) to[out=270,in=0] (park.east);
\end{tikzpicture}
