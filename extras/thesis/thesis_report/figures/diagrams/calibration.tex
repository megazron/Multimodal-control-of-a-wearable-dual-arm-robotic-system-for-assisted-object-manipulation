% CALIBRATION PROCEDURE. Ordered, because each step consumes the previous
% step's output, and two of the orderings are load-bearing.
\begin{tikzpicture}[x=1mm, y=1mm, font=\scriptsize]

\node[proc, text width=36mm] (settle) at (0, 0)
  {\textbf{0. Settle the stream}\\ node running, port held,\\ IMU past power-on};
\node[proc, text width=36mm] (gyro) at (0, -20)
  {\textbf{1. Gyro bias}\\ sample \texttt{/master\_arm\_raw\_*},\\ never the port};
\node[proc, text width=36mm] (zero) at (0, -40)
  {\textbf{2. Zero reference}\\ arm hanging; captures\\ $\hat{a}$ and the j1 datum};
\node[proc, text width=36mm] (chan) at (0, -60)
  {\textbf{3. Channel health}\\ verdict per channel over\\ \emph{distinct} updates};
\node[proc, text width=36mm] (maxp) at (0, -80)
  {\textbf{4. Range}\\ sweep each joint to both\\ mechanical stops};
\node[proc, text width=36mm] (btn) at (0, -100)
  {\textbf{5. Button map}\\ \texttt{check\_buttons}: reports,\\ never rewrites};
\node[actuator, text width=36mm] (anchor) at (0, -120)
  {\textbf{6. Workspace anchor}\\ \texttt{tf2\_echo} at home\\ $\Rightarrow$ $P_{\text{HOME}}$, anchor quaternion};

\foreach \a/\b in {settle/gyro, gyro/zero, zero/chan, chan/maxp, maxp/btn, btn/anchor}
  {\draw[flow] (\a) -- (\b);}

\node[draw, dashed, thick, red!60!black, fill=red!4, text width=64mm,
      align=left, anchor=west] at (30, -30)
  {\textbf{\color{red!60!black}Step 0 is not optional.} Re-opening the serial
   port resets the board, and the IMU emits nonsense while settling. A gyro
   bias captured through a script that reopens the port read
   \SI{4.85}{\degree\per\second} of noise on the left sensor; sampled from the
   settled topic instead, the same gyro read
   \SI{0.11}{\degree\per\second}. That single mistake was the difference
   between \SI{1492}{\degree\per\minute} and \SI{44}{\degree\per\minute} of
   estimated drift.};

\node[draw, dashed, thick, red!60!black, fill=red!4, text width=64mm,
      align=left, anchor=west] at (30, -95)
  {\textbf{\color{red!60!black}Step 3 must precede step 4.} A range sweep on
   an incoherent channel measures the fault, not the joint: the readings span
   a wide interval without forming a trajectory, and range alone calls that
   healthy.};

\node[draw, dashed, thick, red!60!black, fill=red!4, text width=64mm,
      align=left, anchor=west] at (30, -128)
  {\textbf{\color{red!60!black}Step 6 is derived, not chosen.} The anchor is
   the end-effector pose at the home joint angles, so changing the home pose
   or the mount invalidates it and every clearance figure computed from it.};
\end{tikzpicture}
