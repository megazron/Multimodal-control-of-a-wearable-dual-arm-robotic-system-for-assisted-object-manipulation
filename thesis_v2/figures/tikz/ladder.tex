% What the system does as sensor channels are lost.
\begin{tikzpicture}[x=1mm, y=1mm, font=\scriptsize]

  \node[sense, minimum height=9mm, text width=38mm] (raw) at (0,0)
    {Fourteen sensor readings arrive, about fifty times a second};

  \node[choice, below=8mm of raw, text width=30mm] (health)
    {Is this channel behaving?};
  \draw[go] (raw) -- (health);

  \node[stage, right=22mm of health, text width=44mm, minimum height=16mm] (verdict)
    {\textbf{One of four verdicts}\\[2pt]
     \emph{working} \textbullet\ \emph{dropping out} \textbullet\
     \emph{incoherent} (present, fresh and wrong) \textbullet\ \emph{dead}};
  \draw[go] (health) -- (verdict);

  \node[stage, below=9mm of health, text width=44mm, minimum height=12mm] (mode)
    {\textbf{Pick the best mapping the surviving channels allow}, and publish
     which one is in use};
  \draw[go] (health) -- (mode);

  \foreach \i/\lbl/\cost in {
      0/{All shoulder channels working}/{full three-dimensional command},
      1/{One shoulder channel lost}/{costs \SI{63.8}{\milli\metre} of tip accuracy},
      2/{Only one channel left}/{direction only, distance held fixed},
      3/{No usable channel}/{\textbf{refuse: report no position}}}
  {
    \pgfmathsetmacro{\y}{-34 - \i*11}
    \ifnum\i=3
      \node[guard, text width=76mm, minimum height=8mm] at (14,\y) {\lbl \ \textbullet\ \cost};
    \else
      \node[stage, text width=76mm, minimum height=8mm] at (14,\y) {\lbl \ \textbullet\ \cost};
    \fi
  }
  \draw[go] (mode) -- (14,-30);

  \node[tag, text width=86mm, align=left] at (14,-80)
    {The refusal at the bottom is the important rung. On a machine mounted
     beside somebody's head, a position that has been guessed is worse than an
     admitted gap, and the difference has to be visible from outside the
     program.};

\end{tikzpicture}
